```rust
use scraper::{ElementRef, Html, Selector};
use serde::Serialize;

/// One Anime-Countdown entry (`/calendar`, grid view). `air` is unix seconds.
#[derive(Debug, Clone, PartialEq, Serialize)]
pub struct CalendarEntry {
    pub id: u32,
    pub title: String,
    pub episode: u32,
    pub lang: String,
    pub air: u64,
}

fn text(e: ElementRef) -> String {
    e.text().collect::<String>().split_whitespace().collect::<Vec<_>>().join(" ")
}

/// Entries without a title link, a watch link or an air time are skipped.
pub fn parse_calendar(html: &str) -> Vec<CalendarEntry> {
    let doc = Html::parse_document(html);
    let sel = |s: &str| Selector::parse(s).expect("selector");
    let (entry, title, watch, air) = (
        sel("div.calendarEntry"),
        sel(r#"a[data-ajax][href^="/info/"]"#),
        sel(r#"a[href^="/watch/"]"#),
        sel(r#"span[id^="airTime"]"#),
    );
    doc.select(&entry)
        .filter_map(|e| {
            let t = e.select(&title).next()?;
            let id = t.value().attr("href")?.strip_prefix("/info/")?.split(|c: char| !c.is_ascii_digit()).next()?.parse().ok()?;
            // /watch/<id>/<episode>/<lang>#top
            let w = e.select(&watch).next()?.value().attr("href")?;
            let mut parts = w.trim_start_matches("/watch/").split(['/', '#']);
            parts.next()?;
            let episode = parts.next()?.parse().ok()?;
            let lang = parts.next()?.to_string();
            if lang.is_empty() || !lang.bytes().all(|b| b.is_ascii_lowercase()) {
                return None;
            }
            let air = text(e.select(&air).next()?).parse().ok()?;
            Some(CalendarEntry { id, title: text(t), episode, lang, air })
        })
        .collect()
}
```

- [ ] **Step 4: `helpers.ts`** (import type `CalendarEntry` alongside the others from `../api.ts`):

```ts
const dayKey = (d: Date) => `${d.getFullYear()}-${d.getMonth() + 1}-${d.getDate()}`;

/** Calendar rows: entries grouped by local air date and sorted by time; one poster per title and day,
 *  badge "HH:MM · Ep N" (further episodes of the same title that day are appended: "Ep 4, 5"). */
export function calendarDays(entries: CalendarEntry[], now: Date): { key: string; label: string; hits: SearchHit[] }[] {
  const today = dayKey(now), tomorrow = dayKey(new Date(now.getFullYear(), now.getMonth(), now.getDate() + 1));
  const days = new Map<string, { key: string; label: string; hits: SearchHit[] }>();
  for (const e of [...entries].sort((a, b) => a.air - b.air)) {
    const d = new Date(e.air * 1000), key = dayKey(d);
    let day = days.get(key);
    if (!day) {
      const label = key === today ? "Today" : key === tomorrow ? "Tomorrow" : d.toLocaleDateString("en-GB", { weekday: "long", day: "numeric", month: "long" });
      day = { key, label, hits: [] };
      days.set(key, day);
    }
    const same = day.hits.find((h) => h.id === e.id);
    if (same) { same.badge += `, ${e.episode}`; continue; }
    const time = d.toLocaleTimeString("en-GB", { hour: "2-digit", minute: "2-digit" });
    day.hits.push({ id: e.id, title: e.title, genres: [], kind: "", rating: null, badge: `${time} · Ep ${e.episode}` });
  }
  return [...days.values()];
}
```

```svelte
<script lang="ts">
  import { onMount } from "svelte";
  import { api, describeError, type CalendarEntry } from "./api";
  import { calendarDays } from "./ui/helpers";
  import PosterRow from "./ui/PosterRow.svelte";
  let { onopen, onauth }: { onopen: (id: number) => void; onauth: () => void } = $props();
  let entries = $state<CalendarEntry[]>([]), status = $state<"loading" | "ready" | "error">("loading"), error = $state("");
  const days = $derived(calendarDays(entries, new Date()));
  async function load() {
    status = "loading"; error = "";
    try { entries = await api.getCalendar(); status = "ready"; }
    catch (x) { if (x === "NotLoggedIn") onauth(); else { error = describeError(x); status = "error"; } }
  }
  onMount(load);
</script>

{:else}
  {#each days as d (d.key)}<PosterRow title={d.label} hits={d.hits} status="ready" {onopen} onretry={load} />{/each}
{/if}
```
